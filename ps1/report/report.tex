\section{Current with a Lorentzian Density of States}

\subsection{Alternative Form}

Given the following expressions for the number of channel electrons and the current in a nanodevice:
\begin{equation}
    N = \int_{-\infty}^{\infty}
    \frac{\gamma_1 f_1(E) + \gamma_2 f_2(E)}{\gamma_1 + \gamma_2}
    D(E-U)\, \dd{E}
\end{equation}
\begin{equation}
    I = \frac{q}{\hbar}
    \frac{\gamma_1 \gamma_2}{\gamma_1 + \gamma_2}
    \int_{-\infty}^{\infty}
    [f_1(E) - f_2(E)] D(E-U)\, \dd{E}
\end{equation}

These expressions can be rewritten into a form that is more convenient for numerical implementation in MATLAB. First, define a change of variables:
\begin{align*}
	E' &= E - U \\
	E &= E' + U \\
	\dd{E'} &= \dd{E}
\end{align*}

Since subtracting a constant does not alter the limits of integration, the integration bounds remain:
\[
    E' : -\infty \rightarrow \infty
\]

Substituting $E = E' + U$ and relabeling $E' \rightarrow E$ yields:
\begin{equation}\label{eq:num_e2}
    N = \int_{-\infty}^{\infty}
    \frac{\gamma_1 f_1(E+U) + \gamma_2 f_2(E+U)}{\gamma_1 + \gamma_2}
    D(E)\, \dd{E}
\end{equation}
\begin{equation}\label{eq:current2}
    I = \frac{q}{\hbar}
    \frac{\gamma_1 \gamma_2}{\gamma_1 + \gamma_2}
    \int_{-\infty}^{\infty}
    [f_1(E+U) - f_2(E+U)] D(E)\, \dd{E}
\end{equation}

\section{Thermoelectric Current}
Consider a channel with a Lorentzian density of states $D(E)$ given by:
\begin{equation}
    D(E) = D_{\mathcal{E}}(E) = \frac{(\gamma_1 + \gamma_2) / 2\pi}{(E - \mathcal{E})^2 + [(\gamma_1 + \gamma_2) / 2]^2}
\end{equation}
where $\gamma = \gamma_1 + \gamma_2 = \SI{0.01}{eV}$. \subsection{Operation at Absolute Zero}
We consider a system operating at absolute zero temperature ($T = \SI{0}{\kelvin}$). The channel is characterized by a discrete energy level $\mathcal{E}$, described by the density of states function $D(E) = \delta(E - \mathcal{E})$. The coupling coefficients to the source and drain contacts are symmetric and given by:
\[
    \gamma_1 = \gamma_2 = \SI{0.005}{\eV}
\]

\begin{figure}[H]
	\centering
	\includegraphics[width=10cm]{imgs/4a_question_plot.png}
	\caption{Plots provided for the absolute zero system.}
	\label{fig:4a_question_lot}
\end{figure}

\noindent To calculate the current $I$ and the number of channel electrons $N$ for each of these three cases, the difference between $f_1$ and $f_2$ along with the density of states $D(E) = \delta(E-\mathcal{E})$, were analyzed in the context of Equation \ref{eq:num_e2} and \ref{eq:current2}.\\

\noindent For (i) and (iii) it is visibly discernable that $D(E)$ intersects where the Fermi functions are equal, therefore there is no difference in agenda resulting in $I = 0$. However, for (i) the Fermi functions are both at $0$ resulting in \boxed{N = 0}, while for (ii) both are at $1$ resulting in \boxed{N = 1}.\\

\noindent At (ii), since $D(E)$ intersects with the $f2 - f1$, leading to a difference of agenda. Because the difference between $f2 - f1 = 1$, current can be calculated using
\[
    I = \frac{q}{\hbar}\frac{\gamma_1 \gamma_2}{\gamma_1 + \gamma_2} = \boxed{0.608 \unit{\nano\ampere}}
\]

\noindent Since $\gamma_1 = \gamma_2$, the number of channel electrons (N) is simply
\[
    N = \frac{f_1(E+U) + f_2(E+U)}{2} = \boxed{0.5}
\]

\subsection{Nanoscale MOSFET Calculation}
We consider a nanoscale MOSFET characterized by a step-function density of states $D(E)$:
\[
    D(E) =
    \begin{cases}
        10^4 \text{ states/eV} & \text{for } E \ge \SI{-0.1}{\eV} \\
        0 & \text{for } E < \SI{-0.1}{\eV}
    \end{cases}
\]

\noindent The device operates under the bias conditions $V_S = 0$, $V_G = \SI{0.6}{\volt}$, and $V_D = \SI{0.4}{\volt}$. Given these parameters, the self-consistent potential is determined to be $U = \SI{-0.2}{\eV}$, with the equilibrium Fermi level located at $\mu = 0$.\\

\noindent Assuming operation at $T = 0 \unit{\kelvin}$, \verb|problem4c.py| produced the following plots

\begin{figure}[H]
	\centering
	\includegraphics[width=13cm]{imgs/4c_question_plots.png}
	\caption{Plots for the system assuming $T = 0 \unit{\kelvin}$, the shaded area determines the current.}
	\label{fig:4b_question_plot}
\end{figure}

\noindent If $\gamma_1 = \gamma_2 = 0.005 eV$, to calculate the current, first we determine the energy range of the Fermi window at $T=\SI{0}{\kelvin}$. The window extends from the drain potential to the source potential, adjusted by the self-consistent potential $U$:
\begin{align*}
    E_{\text{upper}} &= \mu_1 - U = 0 - (-0.2) = \SI{0.2}{\eV} \\
    E_{\text{lower}} &= \mu_2 - U = -0.4 - (-0.2) = \SI{-0.2}{\eV}
\end{align*}

\noindent The density of states $D(E)$ is non-zero ($10^4$) only for $E \ge \SI{-0.1}{\eV}$. Therefore, the effective integration range is restricted to $[\SI{-0.1}{\eV}, \SI{0.2}{\eV}]$.

\noindent Substituting the given values $\gamma_1 = \gamma_2 = \SI{0.005}{\eV}$ into Equation \ref{eq:current2}:
\begin{align*}
    I &= \frac{q}{\hbar} \frac{\gamma_1 \gamma_2}{\gamma_1 + \gamma_2} \int_{-0.1}^{0.2} D(E) \, \dd{E} \\
    I &= \frac{q}{\hbar} \left( \frac{0.005^2}{0.01} \unit{\eV} \right) \left[ 10^4 \times (0.2 - (-0.1)) \right] \\
    I &= \frac{q}{\hbar} (\SI{0.0025}{\eV}) (3000) \\
    I &= \frac{q}{\hbar} (\SI{7.5}{\eV})
\end{align*}

\noindent Converting $\SI{7.5}{\eV}$ to Joules ($\SI{1.2015e-18}{\joule}$) to solve for the current in Amperes:
\[
    I = \frac{1.602 \times 10^{-19}}{1.055 \times 10^{-34}} \times 1.2015 \times 10^{-18} \approx \SI{1.82}{\milli\ampere}
\]

\noindent At a specified $V_G$ and $V_S$ and a sufficiently high drain voltage, the self-consistent potential saturates to a value of $\SI{-0.3}{\unit{\eV}}$. The current is determined by the area under the curve, and since area changed by a factor of $4/3$ the new current can be determined by multiplying the old current by a factor of $4/3$. This produces a value of \boxed{I = \SI{2.43}{\unit{\milli\ampere}}.}

\subsection{Calculation of Coupling Coefficient $\gamma_2$}

We consider a nanodevice with a density of states approximated by a delta function $D(E) = \delta(E - \mathcal{E})$, where the energy level is located at $\mathcal{E} = \SI{0.2}{\eV}$. The device operates at room temperature ($k_B T = \SI{0.025}{\eV}$) with an equilibrium Fermi level $\mu = 0$.

\noindent The electrical parameters and bias conditions are given as:
\begin{itemize}
    \item Biases: $V_G = V_S = \SI{0}{\volt}$, $V_D = \SI{0.6}{\volt}$.
    \item Capacitance parameters: $\alpha_G = \alpha_D = 0.5$, $\alpha_S = 0$.
    \item Charging energy: $U_0 = \SI{0.25}{\eV}$.
    \item Reference electrons: $N_0 = 0$.
\end{itemize}

\noindent The self-consistent potential $U$ is the sum of the Laplace potential ($U_L$) and the Poisson potential ($U_P$), producing the following equation
\begin{equation}\label{eq:potential}
    U = U_L + U_P = -(\alpha_G V_G + \alpha_S V_S + \alpha_D V_D) + U_0(N - N_0)
\end{equation}
\noindent Given the self-consistent electron number $N = 0.325$ and the source coupling $\gamma_1 = \SI{5.00e-3}{\eV}$, we rearrange Equation \ref{eq:num_e2} to solve for $\gamma_2$
\begin{equation}\label{eq:gamma_1}
    \gamma_2 = \gamma_1 \frac{f_1(\mathcal{E} + U) - N}{N - f_2(\mathcal{E} + U)}
\end{equation}
Using the Equations \ref{eq:potential} and \ref{eq:gamma_1}, first we compute the self-consistent potential $U$:
\begin{align*}
    U &= -[0.5(0) + 0(0) + 0.5(0.6)] + 0.25(0.325 - 0) \\
    U &= -0.3 + 0.08125 = \SI{-0.21875}{\eV}
\end{align*}

\noindent Next, we evaluate the Fermi functions at the shifted energy level $E' = \mathcal{E} + U = 0.2 - 0.21875 = \SI{-0.01875}{\eV}$. With $\mu_1 = 0$ and $\mu_2 = -qV_D = \SI{-0.6}{\eV}$:
\begin{align*}
    f_1 &= \frac{1}{1 + e^{(E' - \mu_1)/k_B T}} = \frac{1}{1 + e^{-0.01875/0.025}} = \frac{1}{1 + e^{-0.75}} \approx 0.6792 \\
    f_2 &= \frac{1}{1 + e^{(E' - \mu_2)/k_B T}} = \frac{1}{1 + e^{(-0.01875 + 0.6)/0.025}} = \frac{1}{1 + e^{23.25}} \approx 0
\end{align*}

\noindent Finally, substituting these values into Equation \ref{eq:gamma_1}:
\[
    \gamma_2 = (5.00 \times 10^{-3}) \frac{0.6792 - 0.325}{0.325 - 0} \approx \boxed{\SI{5.45e-3}{\eV}}
\]

\subsection{Thermoelectric Detector Optimization}
We analyze the thermoelectric current in a single-level channel driven by a temperature difference between the source and drain contacts. The system operates with an equilibrium Fermi level $\mu = 0$, ignoring self-consistency.\\

\noindent The current $I$ through the device is proportional to the difference in the Fermi functions of the two contacts at the molecule's specified energy level $E$. Since $\gamma_1 = \gamma_2$, the difference in agenda can be written as
\begin{equation} \label{eq:delta_f}
    \Delta f(E) = f_1(E) - f_2(E) = \frac{1}{1 + e^{(E - \mu)/k_B T_1}} - \frac{1}{1 + e^{(E - \mu)/k_B T_2}}
\end{equation}

\noindent Given the thermal energies $k_B T_1 = \SI{0.024}{\eV}$ and $k_B T_2 = \SI{0.027}{\eV}$, we evaluate $\Delta f$ for each molecule.\\

\noindent \textbf{Molecule A ($E = \SI{0}{\eV}$):}
\[
    \Delta f_A = \frac{1}{1 + e^0} - \frac{1}{1 + e^0} = 0.5 - 0.5 = 0
\]

\noindent \textbf{Molecule B ($E = \SI{-0.05}{\eV}$):}
\begin{align*}
    f_1 &= \frac{1}{1 + e^{-0.05/0.024}} = \frac{1}{1 + e^{-2.083}} \approx 0.8893 \\
    f_2 &= \frac{1}{1 + e^{-0.05/0.027}} = \frac{1}{1 + e^{-1.852}} \approx 0.8644 \\
    \Delta f_B &= 0.8893 - 0.8644 = 0.0249
\end{align*}

\noindent \textbf{Molecule C ($E = \SI{-0.1}{\eV}$):}
\begin{align*}
    f_1 &= \frac{1}{1 + e^{-0.1/0.024}} = \frac{1}{1 + e^{-4.167}} \approx 0.9847 \\
    f_2 &= \frac{1}{1 + e^{-0.1/0.027}} = \frac{1}{1 + e^{-3.704}} \approx 0.9759 \\
    \Delta f_C &= 0.9847 - 0.9759 = 0.0088
\end{align*}

\noindent Comparing the results, Molecule B yields the largest difference in Fermi distributions. Since the current is proportional to the difference, the thermoelectric current is maximized with \boxed{\textbf{Molecule B}.}

\subsection{Nanoscale MOSFET Channel Analysis}
A nanoscale MOSFET is cooled to $T = \SI{0}{\kelvin}$ to compare two channel materials. The system assumes a self-consistent potential $U = 0$ and an equilibrium Fermi level $\mu = 0$.

\begin{itemize}
    \item Source: $V_S = \SI{-2}{\volt} \implies \mu_1 = \mu - qV_S = \SI{+2}{\eV}$
    \item Drain: $V_D = \SI{1}{\volt} \implies \mu_2 = \mu - qV_D = \SI{-1}{\eV}$
\end{itemize}

\noindent Both materials share a constant density of states $D(E) = 10^4 \, \si{\per\eV}$ and a constant drain coupling $\gamma_2 = \SI{0.005}{\eV}$. The source coupling $\gamma_1$ varies by material:

\noindent \textit{Material A:}
\[
    \gamma_1(E) = \begin{cases} \SI{0.005}{\eV} & E \ge 0 \\ 0 & E < 0 \end{cases}
\]

\noindent \textit{Material B:}
\[
    \gamma_1(E) = \begin{cases} 0 & E \ge 0 \\ \SI{0.005}{\eV} & E < 0 \end{cases}
\]

\noindent The equation of current, Equation \ref{eq:current2}, can be rewritten as.
\[
    I = \frac{q}{\hbar}
    \frac{\gamma_1 \gamma_2}{\gamma_1 + \gamma_2}
    \int_{-\infty}^{\infty}
    [f_1(E) - f_2(E)] D(E)\, \dd{E}
\]

\noindent At $T = \SI{0}{\kelvin}$, the Fermi functions act as step functions defined by $\mu_1 = \SI{2}{\eV}$ and $\mu_2 = \SI{-1}{\eV}$. Their difference is
\begin{equation}
    f_1(E) - f_2(E) =
    \begin{cases}
        1 & \text{for } \mu_2 < E < \mu_1 \ (\text{i.e., } -1 < E < 2) \\
        0 & \text{otherwise}
    \end{cases}
\end{equation}

\subsubsection{Material A}
For Material A, $\gamma_1$ is non-zero only for $E \ge 0$.
\noindent The current integral covers the range $[0, 2]$:
\begin{align*}
    I_A &= \frac{q}{\hbar} \int_{0}^{2} (0.0025)(10^4) \, \dd{E} \\
    I_A &= \frac{q}{\hbar} [25 \times (2 - 0)] = \frac{q}{\hbar} (\SI{50}{\eV}) \approx \boxed{\SI{12.17}{\milli\ampere}}
\end{align*}

\subsubsection{Material B}
For Material B, $\gamma_1$ is non-zero only for $E < 0$.
\noindent The current integral covers the range $[-1, 0]$:
\begin{align*}
    I_B &= \frac{q}{\hbar} \int_{-1}^{0} (0.0025)(10^4) \, \dd{E} \\
    I_B &= \frac{q}{\hbar} [25 \times (0 - (-1))] = \frac{q}{\hbar} (\SI{25}{\eV}) \approx \boxed{\SI{6.09}{\milli\ampere}}
\end{align*}

\subsection{Numbered of Broadened Energy Level}
A small conductor is placed between two contacts and a voltage of $\SI{4.3}{\unit{\milli\volt}}$ is applied. It is found that the maximum current is $\SI{500}{\unit{\nano\ampere}}$. To find the number of broadened energy levels $N$, begin by calculating the conductance
\[
    G = I/V = 1.163 \times 10^{-4}\,\unit{\siemens}
\]
To find $N$ simply divide $G$ by the \textit{quantum of conductance} $G_0$
\[
N = \frac{G}{G_0} = \frac{1.163 \times 10^{-4}\,\unit{\siemens}}{q^2/h}
\]
Now we solve for $N$:
\[
    N = \frac{1.163 \times 10^{-4}}{3.874 \times 10^{-5}} \approx 3.002
\]
So there are \boxed{3} broadened energy levels present in the conductor.
